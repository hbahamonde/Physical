\section{Theory}

\subsection{Political Psychology}

\miniframeson
\begin{frame}[c]{Political Psychology: Heuristics}
    \begin{itemize}
      \item A candidate's {\bf physical appearance} is ``the most important'' {\color{gray}\tiny (Lau \& Redlawsk 2001)} and the ``most obvious and accessible'' {\color{gray}\tiny (Dion et al. 1972)} \emph{{\color{blue}heuristic}} available to voters {\color{gray}\tiny (Stockemer \& Praino 2017)}.\pause

      \item {\bf {\color{blue}Heuristics}} are cognitive devices that allow {\bf reasonable voting decision making with minimal effort} {\color{gray}\tiny (Lau \& Redlawsk 2001 and Tversky \& Kahneman 1973)}.\pause

      \item Thus, ``{\color{blue}voters vote beautiful}'' {\color{gray}\tiny(Efrain \& Patterson 1974)} because attractive candidates ``are more likely to be attributed the {\color{blue}qualities} associated with {\color{blue}successful} politicians'' {\color{gray}\tiny(Stockemer \& Praino 2019)}.

    \end{itemize}
\end{frame}



\subsection{Status Expectations Theory}

\miniframeson
\begin{frame}[c]{Status Expectations Theory}
    \begin{itemize}
      \item {\bf Status Expectations Theory}: {\color{blue}physical attractiveness} is considered a similar status characteristic as gender or occupation.
        \begin{itemize}
          \item[$\checkmark$] For ex., attractive people are `{\color{blue}better}' or `{\color{blue}more capable}' {\tiny\color{gray}(Webster \& Driskell 1983)}.\pause
        \end{itemize}

        \item We extend this concept to {\color{blue}{\bf physical appearance}-based cues of {\bf occupation} as a {\bf status} characteristic}.

    \end{itemize}
\end{frame}

\section{Discussion}

\subsection{At a Glance}

\miniframeson
\begin{frame}[c]{Main Results}
      \begin{itemize}
        \item {\bf Male \& female} candidates:  
          \begin{enumerate}
            %\item[$\checkmark$] Candidates that show {\color{black!30!green}\emph{higher degrees of congruence}} between physical appearance and occupation {\color{black!30!green}do better}.
            \item[$\checkmark$] Candidates that look-like and also perform {\color{black!30!green}\emph{upper-class occupations} do {\bf best}} in elections.\pause
          \end{enumerate}
        \item  {\color{blue}{\bf Female} candidates}:
          \begin{itemize}
          \item Systematic {\color{red}electoral penalty} when they {\color{red}{\bf look-like}} and also {\bf perform {\color{red}working-class} occupations}.
          \item This is represents a 36\%$^{\star\star\star}$ marginal change.\pause
        \end{itemize}
    \item  {\color{blue}{\bf Male} candidates}:
          \begin{itemize}
          \item \emph{Don't} experience the same electoral punishment: it does \emph{not} matter (electorally) if a male candidate looks working-class.
        \end{itemize}

      \end{itemize}
\end{frame}

\subsection{Wrapping Up}

\miniframeson
\begin{frame}[c]{Main Takeaways}
  \begin{itemize}
    \item[$\checkmark$] Considered {\bf the use of heuristics in elections} and the concept of {\bf ``status'' in expectation states theory}.
    \item[$\checkmark$] Selected the {\bf \emph{unlikely} case} of the 2017 Finnish Municipal Elections.
    \item[$\checkmark$] Exploited a {\bf novel representative data set} of candidates' physical appearances.
    \item[$\checkmark$] Proposed an alternative explanation for {\bf voting} not based on attractiveness but on {\bf occupation-appearance congruence} and {\bf social stratification}.
    \item[{\color{red}$\checkmark$}] Female candidates experience a systematic {\color{red}\bf electoral penalty} when they {\color{red}{\bf look}} {\bf {\color{red}working-class}}.
  \end{itemize}
\end{frame}
